若希望完全不開啟郵件軟體、直接於背景自動寄出含 PDF 附件的 Email，可於 Google Apps Script 部署下方簡易 Web App 並貼上網址：

點此展開 10 行 Google Apps Script 自動寄信程式碼 
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var blob = Utilities.newBlob(Utilities.base64Decode(data.pdfBase64), "application/pdf", data.fileName);
  MailApp.sendEmail({
    to: data.to,
    subject: data.subject,
    body: data.body,
    attachments: [blob]
  });
  return ContentService.createTextOutput(JSON.stringify({status:"ok"})).setMimeType(ContentService.MimeType.JSON);
}
